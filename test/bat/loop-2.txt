#!BAT!
@echo off
cls

${a} = "a"
${b} = "b"

echo "aaa"

${i} = 0
LOOP (${i} < 3) (
  echo " loop (i=" + ${i} + ")"

  IF (${a} == "a") (
    echo " a"
  ) ELSE (
    echo " !a"
  )

  ${j} = 0
  LOOP (${j} < 2) (
    echo "  loop (j=" + ${j} + ")"

    IF (${b} == "b") (
      echo "  b"
    ) ELSE (
      echo "  !b"
    )

    ${j}++
  )

  echo ""
  ${i}++
)

echo "bbb"

echo "END"
